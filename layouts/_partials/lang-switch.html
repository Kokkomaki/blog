{{- /* Inline language switcher for a post that has another-language version (see translations.html for the two ways to declare one); renders nothing otherwise. Meant to sit at the end of the .reading-time line. Every language is listed, the current one unlinked. */ -}} {{- $versions := partial "translations.html" . -}} {{- if gt (len $versions) 1 -}} · {{ "" -}} {{- range $i, $v := $versions }} {{- if $i }} / {{ end }} {{- if $v.current }}{{ i18n (printf "lang_%s" $v.lang) }} {{- else }}{{ i18n (printf "lang_%s" $v.lang) }} {{- end }} {{- end }} {{- end -}}
